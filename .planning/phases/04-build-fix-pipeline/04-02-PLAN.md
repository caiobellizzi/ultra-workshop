Create hermes-skills/workshop_push.py — standalone post-approval push+PR script.
This script is called by pr-opener SKILL.md's terminal command AFTER the SKILL.md body has
already received HITL approval (the approval gate lives in the calling SKILL.md body, not here).
- Header: "# Deploy location: /opt/ultra-workshop/hermes-skills/workshop_push.py"
- from __future__ import annotations
- Imports: argparse, os, subprocess, sys, importlib.util; pathlib Path; datetime (datetime, timezone)
- Load brain_http.py via importlib (same try/except pattern as aider_runner.py; store as _brain_http, None if absent)
- REPO = "caiobellizzi/test-workshop-sandbox"
- def main():
    Parse args: --task-id (required str), --branch (required str), --workspace-dir (required str),
                --plan-goal (str, default ""), --diff-summary (str, default "")
    task_id = args.task_id
    branch = args.branch
    workspace_dir = args.workspace_dir

    STEP 1 — git push:
    push_result = subprocess.run(
        ["git", "push", "origin", branch],
        cwd=workspace_dir, capture_output=True, text=True, shell=False,
        env={**os.environ, "GH_TOKEN": os.environ.get("GITHUB_PAT", "")},
    )
    if push_result.returncode != 0:
        print(f"[workshop_push] ERROR: git push failed: {push_result.stderr}", flush=True)
        sys.exit(1)

    STEP 2 — gh pr create:
    pr_title = f"[workshop/{task_id}] {args.plan_goal[:60]}"
    pr_body = f"Auto-generated by ultra-workshop.\n\nTask: {task_id}\n\n{args.diff_summary[:400]}"
    pr_result = subprocess.run(
        ["gh", "pr", "create", "--repo", REPO, "--base", "main", "--head", branch,
         "--title", pr_title, "--body", pr_body],
        capture_output=True, text=True, shell=False,
        env={**os.environ, "GH_TOKEN": os.environ.get("GITHUB_PAT", "")},
    )
    if pr_result.returncode != 0:
        print(f"[workshop_push] ERROR: gh pr create failed: {pr_result.stderr}", flush=True)
        sys.exit(1)
    pr_url = pr_result.stdout.strip()
    print(f"[workshop_push] PR opened: {pr_url}", flush=True)

    STEP 3 — ADR write-back (non-blocking try/except):
    Use EXACT frontmatter template with dotted-namespace keys (per RESEARCH.md §ADR):
    today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    adr_path = f"vault/_system/workshop-adrs/{task_id}.md"
    adr_content = (
        f"---\n"
        f"workshop.task_id: {task_id}\n"
        f"workshop.status: done\n"
        f"workshop.pr_url: {pr_url}\n"
        f"system.created_by: workshop\n"
        f"date: {today}\n"
        f"---\n\n"
        f"# ADR: {task_id}\n\n"
        f"**Goal:** {args.plan_goal}\n\n"
        f"**PR:** {pr_url}\n\n"
        f"{args.diff_summary}\n"
    )
    try:
        if _brain_http:
            _brain_http.call_agent("ingest", f"{adr_path}\n\n{adr_content}")
            print(f"[workshop_push] ADR written: {adr_path}", flush=True)
        else:
            print(f"[workshop_push] WARNING: brain_http not available — ADR not written", file=sys.stderr, flush=True)
    except Exception as exc:
        print(f"[workshop_push] WARNING: ADR write-back failed (non-blocking): {exc}", file=sys.stderr, flush=True)

    print(f"[workshop_push] done — pr_url={pr_url}", flush=True)

- if __name__ == "__main__": main()
